            break;
        }
    }
}

int main(void)
{

    system("pause");
    return EXIT_SUCCESS;

// 管理程序运行的主业务逻辑
#define _CRT_SECURE_NO_WARNINGS

enum {ADD = 1, DEL, SREACH, SHOW, EXIT};

// 显示菜单
void show_menu(void)
{
    printf("\n欢迎使用简易电话簿系统\n\n");
    printf("\t1. 添加联系人\n\n");
    printf("\t2. 删除联系人\n\n");
    printf("\t3. 查找联系人\n\n");
    printf("\t4. 显示联系人\n\n");
    printf("\t5. 退出系统\n\n");
}

// ADD 分支
void add(void)
{
    printf("\n\t【姓名】为：");
    char name[32] = {0};        // 临时获取用户输入，存储联系人姓名
    (void)scanf("%s", name);

    printf("\n\t【电话】为：");
    char phone[16] = {0};       // 临时获取用户输入，存储联系人电话
    (void)scanf("%s", phone);

    // 向联系人数组中添加一个新联系人
    add_person(name, phone);
    printf("\n联系人【%s】添加成功！\n", name);
}

// DEL 分支
void del(void)
{
    printf("\n\t待【删除】的联系人姓名为：");
    char name[32] = {0};
    (void)scanf("%s", name);

    // 从联系人数组中删除联系人
    del_person(name);
    printf("\n联系人【%s】已成功删除！\n", name);
}

// SEARCH 分支
void search(void)
{
    printf("\n\t待【查找】的联系人姓名为：");
    char name[32] = {0};
    (void)scanf("%s", name);

    // 从联系人数组中查找联系人的电话
    char *phone = search_person(name);
    if (NULL == phone)
    {
        printf("\n抱歉！电话簿中【%s】查无此人！\n", name);
    }
    else
    {
        printf("\n联系人【%s】的电话为：%s\n", name, phone);
    }
}

4. 具体功能函数实现

添加联系人

// 具体添加联系人
void add_person(const char *name, const char *phone)
{
    // 校验参数
    if (NULL == name || NULL == phone)
    {
        printf("add_person 添加联系人失败！参数错误！\n");
        return;
    }

    strcpy(numbers[size].name, name);       // 存姓名到数组
    strcpy(numbers[size].phone, phone);     // 存电话到数组
    ++size;                                 // 更新 size
}
要点：1. 先校验参数（指针是否为 NULL）。2. 用 strcpy 拷贝字符串（char[] 是地址常量，不能用 = 赋值）。3. 存完一个就 ++size，size 同时也是"下一个空位的下标"。
