A · 失落的画卷

P1659时间 1 秒 / 内存 128 兆去 OJ 做题 ↗

把一张 n×m 的字符画用 RLE（游程编码）压缩，或把 RLE 串解压回字符画，由 op 决定方向。

思路：压缩：把所有行拼成一个长串，扫一遍数连续相同的字符，写成「次数+字符」；解压：顺序读「数字串+字符」，遇数字就累加计数，遇字符就输出 count 个。

输入
2 5 1
AAAAB
BBCCC
输出
4A3B3C
参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, m, op;
    if (!(cin >> n >> m >> op)) return 0;
    if (op == 1) {
        string s;
        string line;
        getline(cin, line);                 // 吃掉行尾
        for (int i = 0; i < n; ++i) {
            getline(cin, line);
            s += line;
        }
        string out;
        for (size_t i = 0; i < s.size(); ) {
            size_t j = i;
            while (j < s.size() && s[j] == s[i]) ++j;
            out += to_string(j - i);
            out += s[i];
            i = j;
        }
        cout << out << '\n';
    } else {
        string s;
        cin >> s;
        string out;
        int cnt = 0;
        for (char c : s) {
            if (isdigit((unsigned char)c)) cnt = cnt * 10 + (c - '0');
            else { out.append(cnt, c); cnt = 0; }
        }
        for (int i = 0; i < n; ++i) {
            cout << out.substr(1LL * i * m, m) << '\n';
        }
    }
    return 0;
}

复杂度：O(nm)
坑点：n 或 m 为 0 时字符画为空，什么都不能输出；次数可能超过 9，解析时要按十进制累加。

B · 坏掉的时针

P1660时间 2 秒 / 内存 32 兆去 OJ 做题 ↗

f(x) = 各位数字 +1 之积，对 n 连续做 x 次，求结果（n ≤ 1e18，x ≤ 1e9，T ≤ 2e6）。

思路：一次操作后结果只含 2/3/5/7 这些因子，再走几步必然掉进 2→3→…→10→2 的 9 长环。预处理 1e6 以内每个数「到环的步数、进环入口」，更大的数直接迭代到 ≤1e6 再查表；超过环前步数就用 (x−dep) mod 9 定位。

输入
2
999999 1
20250927 2
输出
1000000
420
参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;

static const int LIM = 1000000;
static unsigned char dep[LIM + 1];
static unsigned char ent[LIM + 1];
static int F[LIM + 1];

static inline long long f(long long v) {
    long long r = 1;
    if (v == 0) return 1;
    while (v > 0) { r *= (v % 10) + 1; v /= 10; }
    return r;
}

int main() {
    for (int v = 0; v <= LIM; ++v) F[v] = (int)f(v);

    memset(dep, 0xFF, sizeof(dep));          // 0xFF = 未知
    memset(ent, 0xFF, sizeof(ent));
    for (int v = 2; v <= 10; ++v) { dep[v] = 0; ent[v] = (unsigned char)v; }

    static int path[64];
    for (int v = 0; v <= LIM; ++v) {
        if (dep[v] != 0xFF) continue;
        int len = 0, cur = v;
        while (dep[cur] == 0xFF && len < 60) { path[len++] = cur; cur = F[cur]; }
        unsigned char d = dep[cur], e = ent[cur];
        for (int i = len - 1; i >= 0; --i) {
            if (++d > 250) d = 250;
            dep[path[i]] = d;
            ent[path[i]] = e;
        }
    }

    static char buf[1 << 25];
    size_t len = fread(buf, 1, sizeof(buf), stdin);
    size_t p = 0;
    auto readLL = [&]() -> long long {
        while (p < len && (buf[p] < '0' || buf[p] > '9')) ++p;
        long long x = 0;
        while (p < len && buf[p] >= '0' && buf[p] <= '9') x = x * 10 + (buf[p++] - '0');
        return x;
    };

    long long T = readLL();
    string out;
    out.reserve(T * 8);
    while (T--) {
        long long n = readLL(), x = readLL();
        if (x == 0) { out += to_string(n); out += '\n'; continue; }
        long long v = f(n);
        --x;
        while (x > 0 && v > LIM) { v = f(v); --x; }
        if (x == 0) { out += to_string(v); out += '\n'; continue; }
        if (x < dep[v]) {                     // 还没进环，直接走
            while (x-- > 0) v = f(v);
            out += to_string(v); out += '\n';
            continue;
        }
        long long t = x - dep[v];             // 进环后再走 t 步
        int e = ent[v];
        int idx = (e - 2 + t % 9) % 9;        // 环：2,3,...,10
        out += to_string(2 + idx); out += '\n';
    }
    fwrite(out.data(), 1, out.size(), stdout);
    return 0;
}

复杂度：预处理 O(LIM·log)；单次询问 O(1)（大数多走常数步）
坑点：T 高达 2e6，必须自己写缓冲读入；x=0 时原样输出 n。

C · CS2: 图神的伤害计算器

P1661时间 1 秒 / 内存 256 兆去 OJ 做题 ↗

三种战斗场景：合计伤害、护甲减伤、连发总伤害，按 option 分别做 a+b、a−b、a×b。

思路：读入 option 后直接分支即可，是热身题。

输入
3
1 100 800
2 2000 500
3 150 4
输出
900
1500
600
参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        int op; long long a, b;
        scanf("%d %lld %lld", &op, &a, &b);
        long long ans = (op == 1 ? a + b : op == 2 ? a - b : a * b);
        printf("%lld\n", ans);
    }
    return 0;
}

复杂度：O(1)
坑点：a,b ≤ 2e9，乘积到 4e18，必须用 long long（scanf 用 %lld）。

D · 奶龙数日子

P1662时间 1 秒 / 内存 256 兆去 OJ 做题 ↗

给两个合法日期，求相差多少天（起点那天算在内、终点不算）。

思路：写 idx(y,m,d) 把日期映射成「从 1800-01-01 起过了多少天」：年按闰年规则累加 365/366，月按表累加，二月在闰年 +1；答案就是 idx2 − idx1。

输入
4
2025 9 18 2025 9 20
2024 2 25 2024 3 5
2023 12 30 2024 1 2
2023 1 1 2025 1 1
输出
2
9
3
731
参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
static bool leap(int y) { return (y % 4 == 0 && y % 100 != 0) || y % 400 == 0; }
static long long idx(int y, int m, int d) {
    static const int md[] = {0,31,28,31,30,31,30,31,31,30,31,30,31};
    long long r = d - 1;
    for (int i = 1; i < m; ++i) r += md[i] + (i == 2 && leap(y));
    for (int i = 1800; i < y; ++i) r += leap(i) ? 366 : 365;
    return r;
}
int main() {
    int t; scanf("%d", &t);
    while (t--) {
        int y1, m1, d1, y2, m2, d2;
        scanf("%d %d %d %d %d %d", &y1, &m1, &d1, &y2, &m2, &d2);
        printf("%lld\n", idx(y2, m2, d2) - idx(y1, m1, d1));
    }
    return 0;
}

复杂度：O(年数) 每年常数，实际近 O(1)
坑点：闰年规则是 4 的倍数、不是 100 的倍数，或 400 的倍数；跨年样例（2023→2025）要能对上 731。

F · 小孤独买拨片


P1664时间 1 秒 / 内存 256 兆去 OJ 做题 ↗
拨片有 50 枚 130 元、10 枚 28 元、1 枚 3 元三种买法，买恰好 n 枚的最小花费（n ≤ 1e5）。


思路：枚举大包个数 a（0..n/50），剩下的 r < 50 再枚举中包个数 b（0..r/10），其余单买，取最小。


输入
4
13
63
100
528
输出
37
167
260
1380


参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
int main() {
    int t; scanf("%d", &t);
    while (t--) {
        int n; scanf("%d", &n);
        long long best = LLONG_MAX;
        for (int a = 0; a * 50 <= n; ++a) {
            int r = n - a * 50;
            for (int b = 0; b * 10 <= r; ++b) {
                int c = r - b * 10;
                best = min(best, 1LL * a * 130 + 1LL * b * 28 + 1LL * c * 3);
            }
        }
        printf("%lld\n", best);
    }
    return 0;
}


复杂度：O((n/50)·5) 每组
坑点：不能「多买」凑整：必须恰好 n 枚；n<10 时单买可能比中包便宜（8 枚 24 < 28）。


G · 共鸣圣歌


P1665时间 1 秒 / 内存 256 兆去 OJ 做题 ↗
在数轴上选一个点 j 站着，收集 Σ max(0, sᵢ − |pᵢ − pⱼ|)，求最大值（N ≤ 1000）。


思路：直接枚举停留点 j，内层扫所有 i 累加贡献，取最大。


输入
3
10 20
20 5
22 12
输出
25


参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n; scanf("%d", &n);
    vector<int> p(n), s(n);
    for (int i = 0; i < n; ++i) scanf("%d %d", &p[i], &s[i]);
    long long best = 0;
    for (int j = 0; j < n; ++j) {
        long long sum = 0;
        for (int i = 0; i < n; ++i) {
            int d = abs(p[i] - p[j]);
            if (d < s[i]) sum += s[i] - d;
        }
        best = max(best, sum);
    }
    printf("%lld\n", best);
    return 0;
}


复杂度：O(N²)
坑点：贡献下界是 0，距离超过 sᵢ 要按 0 算，不能出现负数。


H · AC is all you need


P1666时间 1 秒 / 内存 128 兆去 OJ 做题 ↗
有 A、B、C 三种字母和 X∈{A,B}、Y∈{A,C}、Z∈{B,C}，拼 ACA/ACB/ACC，求最多拼几个。


思路：一个串 = 「首字母 A」+「中间 C」+「末尾任意」。二分答案 k，建图跑最大流判可行：源点连 6 种字母（容量为个数），字母连它能担任的角色（首/A、中/C、尾/任意），角色连汇点容量 k，满流 3k 即可行。


输入
5
1 1 4 5 1 4
6 1 3 0 0 0
3 3 3 3 3 3
5 9 1 8 7 2
111111111 222222222 333333333 666666666 777777777 888888888
输出
5
3
6
10
999999999


参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

struct Dinic {
    struct E { int to, rev; ll cap; };
    vector<vector<E>> g;
    vector<int> lv, it;
    Dinic(int n) : g(n), lv(n), it(n) {}
    void add(int u, int v, ll c) {
        g[u].push_back({v, (int)g[v].size(), c});
        g[v].push_back({u, (int)g[u].size() - 1, 0});
    }
    bool bfs(int s, int t) {
        fill(lv.begin(), lv.end(), -1);
        queue<int> q; lv[s] = 0; q.push(s);
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (auto &e : g[u]) if (e.cap > 0 && lv[e.to] < 0) { lv[e.to] = lv[u] + 1; q.push(e.to); }
        }
        return lv[t] >= 0;
    }
    ll dfs(int u, int t, ll f) {
        if (u == t) return f;
        for (int &i = it[u]; i < (int)g[u].size(); ++i) {
            E &e = g[u][i];
            if (e.cap > 0 && lv[e.to] == lv[u] + 1) {
                ll d = dfs(e.to, t, min(f, e.cap));
                if (d > 0) { e.cap -= d; g[e.to][e.rev].cap += d; return d; }
            }
        }
        return 0;
    }
    ll maxflow(int s, int t) {
        ll f = 0;
        while (bfs(s, t)) { fill(it.begin(), it.end(), 0); while (ll d = dfs(s, t, LLONG_MAX)) f += d; }
        return f;
    }
};

int main() {
    int t; scanf("%d", &t);
    while (t--) {
        ll a, b, c, x, y, z;
        scanf("%lld %lld %lld %lld %lld %lld", &a, &b, &c, &x, &y, &z);
        // 角色：0 = 首字母 A；1 = 中间 C；2 = 末尾任意
        // 字母：A B C X Y Z
        ll cnt[6] = {a, b, c, x, y, z};
        bool ok[6][3] = {
            {1, 0, 1},  // A: A 首 / 任意
            {0, 0, 1},  // B: 任意
            {0, 1, 1},  // C: C 中 / 任意
            {1, 0, 1},  // X: A 首 / 任意
            {1, 1, 1},  // Y: A 首 / C 中 / 任意
            {0, 1, 1}   // Z: C 中 / 任意
        };
        auto feasible = [&](ll k) -> bool {
            if (k == 0) return true;
            int S = 9, T = 10;
            Dinic din(11);
            for (int i = 0; i < 6; ++i) din.add(S, i, cnt[i]);
            for (int i = 0; i < 6; ++i)
                for (int r = 0; r < 3; ++r)
                    if (ok[i][r]) din.add(i, 6 + r, LLONG_MAX / 4);
            for (int r = 0; r < 3; ++r) din.add(6 + r, T, k);
            return din.maxflow(S, T) == 3 * k;
        };
        ll lo = 0, hi = (a + b + c + x + y + z) / 3, ans = 0;
        while (lo <= hi) {
            ll mid = (lo + hi) / 2;
            if (feasible(mid)) { ans = mid; lo = mid + 1; } else hi = mid - 1;
        }
        printf("%lld\n", ans);
    }
    return 0;
}


复杂度：O(log·最大流)，图只有 11 个点
坑点：Y 既能当 A 又能当 C，必须靠流来分配，不能简单取 min；答案可到 1e9，容量用 long long。


I · 信息摘要


P1667时间 1 秒 / 内存 128 兆去 OJ 做题 ↗
长度为 n 的分数序列，多次询问区间 [l,r] 的乘积 mod 998244353（即 P·Q⁻¹）。


思路：每个分数先化成 p·q⁻¹ mod M，做前缀积；区间答案 = pre[r] · pre[l−1]⁻¹，逆元用费马小定理快速幂。


输入
3
2 1
1 4
1 5
2
1 1
2 3
输出
2
149736653


参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353;
ll qpow(ll a, ll b) { ll r = 1; a %= MOD; while (b) { if (b & 1) r = r * a % MOD; a = a * a % MOD; b >>= 1; } return r; }
int main() {
    int n; scanf("%d", &n);
    vector<ll> pre(n + 1, 1);
    for (int i = 1; i <= n; ++i) {
        ll p, q; scanf("%lld %lld", &p, &q);
        pre[i] = pre[i - 1] * (p % MOD) % MOD * qpow(q % MOD, MOD - 2) % MOD;
    }
    int m; scanf("%d", &m);
    while (m--) {
        int l, r; scanf("%d %d", &l, &r);
        ll ans = pre[r] * qpow(pre[l - 1], MOD - 2) % MOD;
        printf("%lld\n", ans);
    }
    return 0;
}


复杂度：O((n+m) log MOD)
坑点：分母要 mod 后再求逆；注意读入可能很大，用 long long 取模。


J · 芙莉莲与世代的魔导书


P1668时间 1 秒 / 内存 128 兆去 OJ 做题 ↗
偶数页 N 的书，每次「织法」把前后两半交错，问做 M 次后第 L 页原来是第几页（N,M ≤ 1e10）。


思路：一次织法相当于把页码 x 变成 x·2⁻¹ (mod N+1)：验证 N=6 时 1→4、2→1、3→5、4→2、5→6、6→3 都对。于是答案是 L·(2⁻¹)^M mod (N+1)，2⁻¹ = (N+2)/2，快速幂即可。


输入
6 2 3
输出
6


参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef __int128 lll;

int main() {
    ll N, M, L;
    scanf("%lld %lld %lld", &N, &M, &L);
    ll mod = N + 1;                    // s(x) = x * inv2 (mod N+1)
    ll inv2 = (N + 2) / 2;             // 2 * inv2 ≡ 1 (mod N+1)
    auto mul = [&](ll a, ll b) { return (ll)((lll)a * b % mod); };
    ll pw = 1, base = inv2 % mod, e = M;
    while (e) { if (e & 1) pw = mul(pw, base); base = mul(base, base); e >>= 1; }
    ll ans = mul(L % mod, pw);
    if (ans == 0) ans = mod;
    printf("%lld\n", ans);
    return 0;
}


复杂度：O(log M)
坑点：N 到 1e10，模乘要用 __int128 防溢出；M=0 时答案就是 L。


K · 九宫格(Easy Version)


P1669时间 2 秒 / 内存 512 兆去 OJ 做题 ↗
n×m（≤3×3）点阵上数所有合法解锁路径：跨点时中间点必须已被访问，单点也算一条。


思路：点数最多 9，直接 DFS 枚举所有简单路径；每一步用 gcd(|dx|,|dy|) 判断两点之间有没有格点，有就检查它们是否都已在路径上。


输入
1 3
输出
11


参考代码（C++17，已过样例）
#include <bits/stdc++.h>
using namespace std;
int n, m, N;
long long ans = 0;
int vis[9];

int g(int x) { return x < 0 ? -x : x; }

bool between_ok(int a, int b) {              // a→b 中间的点是否都已访问
    int ax = a / m, ay = a % m, bx = b / m, by = b % m;
    int dx = bx - ax, dy = by - ay;
    int gg = __gcd(g(dx), g(dy));
    if (gg <= 1) return true;                // 相邻，中间无点
    int sx = dx / gg, sy = dy / gg;
    for (int k = 1; k < gg; ++k) {
        int x = ax + sx * k, y = ay + sy * k;
        if (!vis[x * m + y]) return false;
    }
    return true;
}

void dfs(int cur, int cnt) {
    ans += 1;                                 // 当前路径本身算一条
    if (cnt == N) return;
    for (int nx = 0; nx < N; ++nx) {
        if (vis[nx]) continue;
        if (!between_ok(cur, nx)) continue;
        vis[nx] = 1;
        dfs(nx, cnt + 1);
        vis[nx] = 0;
    }
}

int main() {
    scanf("%d %d", &n, &m);
    N = n * m;
    for (int s = 0; s < N; ++s) {
        vis[s] = 1;
        dfs(s, 1);
        vis[s] = 0;
    }
    printf("%lld\n", ans);
    return 0;
}


复杂度：O(所有简单路径) ≈ 1e6 量级
坑点：1×3 的答案是 11 不是 9：B→A→C 这类「先经过中点再跨」是合法的；路径长度 1 也要计数。

思路：经典 Bash 博弈：3 的倍数时后手必胜，否则先手取 n mod 3 块把局面留成 3 的倍数。

4


输出
Yui
