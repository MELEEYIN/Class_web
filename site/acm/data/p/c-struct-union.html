1. Heap 空间操作：二级指针对应的 heap 空间

int **p  ==  int *p[3]  ==>  [int *, int *, int *]  ==>  [[1,2,3,54,5], int *, int *]
int main(void)
{
    // 给外层空间 malloc 申请内存
    int **p = malloc(sizeof(int *) * 3);
    if (p == NULL)
    {
        printf("malloc error!\n");
        return -1;
    }

    // 给内层指针申请 malloc 空间
    for (size_t i = 0; i < 3; i++)
    {
        p[i] = malloc(sizeof(int) * 5);
        if (p[i] == NULL)
        {
            printf("malloc p[i] error!\n");
            return -1;
        }
    }

    // 使用空间 —— 写
    for (size_t i = 0; i < 3; i++)
    {
        for (size_t j = 0; j < 5; j++)
        {
            p[i][j] = i + j;    // 随意初始化值
        }
    }

    // 使用空间 —— 读
    for (size_t i = 0; i < 3; i++)
    {
        for (size_t j = 0; j < 5; j++)
        {
            printf("%d ", *(*(p + i) + j));     // p[i][j] == *(*(p+i)+j)
        }
        printf("\n");
    }

    // free 空间时：应该先释放内层空间，再释放外层
    for (size_t i = 0; i < 3; i++)
    {
        free(p[i]);
        p[i] = NULL;
    }
    // 释放外层空间
    free(p);
    p = NULL;

    system("pause");
    return EXIT_SUCCESS;
}

以 char **p 为例

// 先申请外层指针
char **p = malloc(sizeof(char *) * 5);

// 申请内层指针
for (int i = 0; i < 5; i++)
{
    p[i] = malloc(sizeof(char) * 10);   // 字符串长度 <= 10 个字符
}

// 写
for (int i = 0; i < 5; i++)
{
    // p[i] = "hello";  错误！
    strcpy(p[i], "hello");
}

// 释放内层
for (int i = 0; i < 5; i++)
{
    free(p[i]);
    p[i] = NULL;
}

// 释放外层
free(p);
p = NULL;
记忆点：申请时先外层后内层，释放时先内层后外层。

2. 内存操作函数（补充）

以下 4 个函数专门用来操作 heap 内存。stack 由系统自动申请、自动释放。基础用法见第 09 章，这里补充细节。

memset

绝大多数情况下，memset 用来将申请好的 heap 内存置 0 —— 单位：字节。

  • memset 函数按"字节"设置。
  • 置 0 ——> 每一个字节都为 0。
  • 【了解】置 1 ——> 每一个字节都为 1 —— 4 字节 == 0x01010101 == 16843009
#include <string.h>
void *memset(void *s, int c, size_t n);

// 参1：内存首地址
// 参2：置成什么，一般传 0
// 参3：内存大小，单位：字节
// 返回值：成功 —— 设置后的地址；失败 —— NULL

// 常用写法
memset(首地址, 0, 空间大小);
int *p = (int *)malloc(sizeof(int) * 10);
if (p == NULL)
{
    printf("malloc error");
    return -1;
}

// 将申请好的内存，全部置 0
memset(p, 0, sizeof(int) * 10);

// 直接打印申请好的空间内容
for (size_t i = 0; i < 10; i++)
{
    printf("%d ", p[i]);
}

free(p);
p = NULL;

memcpy

以字节为单元做内存拷贝。strcpy 只能拷贝字符串，memcpy 什么都能拷。

void *memcpy(void *dest, const void *src, size_t n);
// 拷贝内存中的整型数据
int arr[10] = {1, 2, 3, 4, 5};
int arr2[10];
memcpy(arr2, arr, sizeof(int) * 10);
for (size_t i = 0; i < 10; i++)
{
    printf("%d ", arr2[i]);
}
printf("\n");

// 拷贝内存中的字符串
char str[] = "hello world";
char p[100];
// memcpy(p, str, strlen(str) + 1);    // 按内存拷贝
strcpy(p, str);                        // 字符串拷贝
strncpy(p, str, strlen(str) + 1);
printf("p = %s\n", p);

memmove

作用完全等同于 memcpy：以字节为单元做内存拷贝，但是安全的！

如果 src 和 dest 之间有重叠，memcpy 由于底层实现原因有可能出错，推荐使用 memmove。
void *memmove(void *dest, const void *src, size_t n);

memcmp

以字节为单位比较内存！作用可以完全参照 strncmp。

规则：s1 == s2 ——> 0；s1 > s2 ——> 1；s1 < s2 ——> -1

int memcmp(const void *s1, const void *s2, size_t n);
int arr1[] = {14, 56, 22, 59, 21, 49, 25, 86, 89, 66, 71, 31, 98};
int arr2[] = {14, 56, 29, 59, 21};

int ret = memcmp(arr1, arr2, 5 * sizeof(int));
printf("ret = %d\n", ret);

3. 内存常见问题（重点）

1）申请 0 字节空间

C 语言中允许申请 0 字节的内存空间，但是 0 字节的空间不能拿来使用。

int *p = malloc(0);
free(p);  p = NULL;   // 可以 free，但不应该使用

2）free 空指针

  • 空指针（NULL）反复 free 不会报错。
  • 非空指针反复 free 会报错！！！
  • —— 推荐 free 后的指针一定置 NULL。

3）越界访问

不允许！！！会导致程序崩溃。

4）free ++ 后的地址

不能正常释放（因为 free 要求的是 malloc 返回的原始地址）。

如果程序中必须要使用 p++，请定义临时变量保存 p 的原始值，以便 free 释放。

4. 函数内申请空间、函数外使用

模式一：A 函数内申请，A 函数使用

注意上述 4 点即可。

int *p = malloc(10);
free(p);  p = NULL;

模式二：A 函数申请空间，B 函数使用（返回指针）

// A 函数申请空间，B 函数使用
int *func1(void)
{
    int *p = malloc(4);
    *p = 456;
    return p;       // 返回的是 heap 堆空间的地址值，函数调用结束，地址依然有效

    // return &p;   // 返回 stack 栈空间的地址值，函数调用结束，地址失效！
}

int main(void)
{
    int *ret = NULL;
    ret = func1();
    printf("%d\n", *ret);

    free(ret);
    ret = NULL;

    system("pause");
    return EXIT_SUCCESS;
}

模式二（变体）：通过参数返回

// A 函数申请空间，B 函数使用
int *func2(int *p)
{
    p = malloc(4);
    return p;
}

int main(void)
{
    int *ret = NULL;
    ret = func2(ret);
    *ret = 345;             // 写
    printf("%d\n", *ret);   // 读

    free(ret);
    ret = NULL;

    system("pause");
    return EXIT_SUCCESS;
}

模式三：传二级指针（推荐）

void func3(int **p)     // int ** 是一级指针的地址
{
    *p = malloc(4);
}

int main(void)
{
    int *ret = NULL;
    func3(&ret);        // 传一级指针的地址

    // func3 函数调用完成后，ret 指针不再为 NULL，而是指向一块有效的 heap 空间
    *ret = 789;
    printf("%d\n", *ret);

    free(ret);
    ret = NULL;

    system("pause");
    return EXIT_SUCCESS;
}

5. 结构体

结构体定义语法

  • 复合类型（用户自定义类型）：int []、int *、char **、struct student
  • 以上定义了一个结构体类型，名字叫 struct student。
  • struct student 的地位等同于 int、char、short、char *、int[]、long long。
  • 通常结构体类型定义在全局位置，或者放到 xxx.h 头文件。
// 定义结构体类型
struct student {
    int age;        // 成员变量 —— 属性。不能被赋初值
    int num;
    char name[10];
};

// 用结构体类型定义变量
struct student a;   // 定义了一个结构体类型的变量 a

头文件写法

// 头文件守卫
#ifndef _XXX_H_
#define _XXX_H_

// 4 种内容：#include、宏定义、函数声明、类型定义（结构体类型）

#endif

普通结构体变量

struct student stu1, stu2, stu3;            // 一次定义 3 个变量，没赋初值
struct student stu = {18, 1, "Andy"};       // 定义并赋初值

访问成员方法

使用 . 访问成员。

// 定义一个结构体变量，赋初值
struct student stu = {18, 1, "Andy"};
printf("first：age = %d, name = %s, num = %d\n", stu.age, stu.name, stu.num);

stu.age = 118;
stu.num = 119;
// stu.name = "cuihua";     // name 为地址常量，不能被赋值
strcpy(stu.name, "cuihua");

printf("last：age = %d, name = %s, num = %d\n", stu.age, stu.name, stu.num);
// 普通变量使用 "->" 访问成员 —— 不常用！
(&stu)->age = 118;
(&stu)->num = 119;

结构体指针变量

struct student *p1, *p2, *p3;        // 一次定义 3 个指针变量 —— 野指针！！！

struct student stu, *p1;             // 一次定义两个：普通变量 stu 和指针变量 p1
p1 = &stu;                           // 给指针初始化
// 或者指向 heap 空间
struct student *p1;
p1 = (struct student *)malloc(sizeof(struct student));

访问成员方法

使用 -> 访问成员。

(*p1).age = 18;
strcpy((*p1).name, "cuihua");
(*p1).num = 119;
注意避免野指针、空指针：1. 指针必须初始化（指向有效变量或 heap 空间）。2. 使用 -> 之前先判断指针是否为 NULL。指针使用 . 访问成员 —— 不常用（需先 (*p) 解引用）。

非常规定义语法（了解）

// 定义结构体类型的同时，定义 1 个结构体变量 s1，一个指针变量 s2
struct student {
    int age;
    int num;
    char name[10];
} s1, *s2;

// 匿名结构体：定义类型的同时定义 1 个结构体变量 s3，一个指针变量 s4
struct {
    int age;
    int num;
    char name[10];
} s3, *s4;      // 无法再定义其他变量

结构体数组

struct student {

struct student stu[5] = {{18, 1, "Andy"}, {19, 2, "Lucy"}, {118, 3, "李四"}};

结构体嵌套

struct person {
    int age;
    char name[10];
};  // 类型

struct student {
    struct person man;      // person 类型的变量，作为 student 类型的成员
    int id;
    char addr[100];
};

int main(void)
{
    struct student stu = {{18, "zhaoliu"}, 1, "北京朝阳区"};

    printf("age = %d\n", stu.man.age);

    strcpy(stu.man.name, "张三丰");
    strcpy(stu.addr, "武当山");

    printf("\nage = %d\n", stu.man.age);
    printf("name = %s\n", stu.man.name);
    printf("addr = %s\n", stu.addr);

    system("pause");
    return EXIT_SUCCESS;
}

结构体变量赋值

主要应用于函数调用期间，实参给形参赋值。

要求：结构体变量赋值时，必须类型相同、成员个数一致、顺序一致。

做函数参数、返回值

方式说明
传值结构体变量做函数参数，将结构体变量的值（实参）拷贝一份给形参，形参、实参共 2 份结构体
传址结构体指针变量做函数参数，将结构体的地址值做实参拷贝一份给形参，形参、实参共 1 份结构体
结论：结构体做函数参数、返回值时，通常采用"传址"方式，节省空间。

含有指针成员的结构体

struct student {
    int age;
    int num;
    char *name;     // 野指针
};

int main(void)
{
    struct student *p;      // 野指针

    // 给 p 初始化堆空间
    p = malloc(sizeof(struct student));
    if (NULL == p)
    {
        printf("malloc p error\n");
        return -1;
    }

    // 给成员变量 name 开辟堆空间
    p->name = malloc(sizeof(char) * 100);
    if (NULL == p->name)
    {
        printf("malloc p->name error\n");
        return -1;
    }

    // 写数据到结构体中
    p->age = 100;
    p->num = 10;
    strcpy(p->name, "张三丰");
    printf("age=%d, name=%s, num=%d\n", p->age, p->name, p->num);

    // 先释放内层空间
    free(p->name);
    p->name = NULL;

    // 再释放外层
    free(p);
    p = NULL;

    return 0;
}
记忆点：申请内存：先申请外层空间，再申请内层空间；释放内存：先释放内层空间，再释放外层空间。

用二级指针给结构体指针分配空间

void func08(struct student **m)
{
    *m = malloc(sizeof(struct student));
    if (NULL == *m)
    {
        printf("malloc error\n");
        return -1;
    }

    (*m)->age = 100;
    (*m)->num = 1;
    strcpy((*m)->name, "zyx");
}

int main(void)
{
    struct student *p = NULL;   // 空指针
    func08(&p);

    printf("age=%d, name=%s, num=%d\n", p->age, p->name, p->num);

    free(p);
    p = NULL;

    system("pause");
    return EXIT_SUCCESS;
}

6. typedef 关键字

作用：给现有的数据类型起别名。【注意】：不能定义新数据类型。

通常使用 typedef 定义过的类型，添加一个 _t 结尾。

定义语法：typedef 旧类型名 新类型名 _t;

typedef unsigned int size_t;    // 给 unsigned int 起别名叫 size_t

int a;              // a 是变量名
typedef int a;      // a 变成了类型名
a b;                // 定义一个整型变量 b（可读性差，不推荐这样用）
typedef struct student {
    int age;
    int num;
    char *name;     // 野指针
} stu_t;            // 新类型名：stu_t

// 定义变量
struct student stu1;    // 依然可以正常使用
stu_t stu2;             // 定义一个 struct student 类型的变量
typedef long long int32_t;  // int 《----》 long long

struct student {
    int age;
    int32_t num;
    char *name;     // 野指针
    int32_t num1;
    int32_t num2;
    int32_t num3;
    int32_t num4;
} stu_t;

使用 typedef 的好处

  1. 简化类型名。
  2. 便于代码的修改和维护。

7. 共用体（联合体）

特性

  1. 内部所有成员变量的地址一致，等同于整个联合体的地址。
  2. 联合体的大小，是内部成员变量中最大的那个成员的大小（对齐）。
  3. 修改其中一个成员的值，其他成员的值也跟着变化。
union test {
    char ch;
    short sh;
    int var;
};  // 创建一个联合体类型
typedef union test {
    char ch;
    short sh;
    int var;
} test_t;

int main(void)
{
    test_t obj;
    obj.var = 0x87654321;

    printf("&obj    = %p\n", &obj);
    printf("&obj.ch = %p\n", &obj.ch);
    printf("&obj.sh = %p\n", &obj.sh);
    printf("&obj.var= %p\n", &obj.var);

    printf("sizeof(test_t) = %u\n", sizeof(test_t));

    printf("var = %#x\n", obj.var);
    printf("sh  = %#x\n", obj.sh);
    printf("ch  = %#x\n", obj.ch);

    obj.ch = 0xAA;                  // 只改了一个字节
    printf("var = %#x\n", obj.var); // 其他成员也跟着变
    printf("sh  = %#x\n", obj.sh);
    printf("ch  = %#x\n", obj.ch);

    system("pause");
    return EXIT_SUCCESS;
}
小端序机器上可以观察到：obj.ch = 0xAA 之后，obj.var 的最低字节变成了 0xAA（即 0x876543AA），说明三个成员共用同一块内存。
